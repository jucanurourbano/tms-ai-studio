# === SQL (Agente BD: render y validación de DDL sin BD ni red) ===
# sqlglot parsea/transpila SQL en el dialecto destino (postgresql, sqlserver,
# oracle, mysql) en Python puro: valida la sintaxis del DDL generado sin levantar
# ningún motor. Pinneado como el resto del stack: un cambio de versión puede
# alterar el parseo de un dialecto y romper la validación en silencio.
sqlglot==30.15.0

# === OpenAPI (Agente API: validación de la especificación sin red ni tooling) ===
# openapi-spec-validator valida el documento generado contra el JSON Schema de
# OpenAPI 3.1 en Python puro. Es a este agente lo que sqlglot es al Agente BD: la
# capa que demuestra que la salida ejecutable es válida, sin depender de una
# herramienta externa. Pinneado por el mismo motivo: un cambio de versión puede
# alterar la validación en silencio.
#
# 0.8.5 y NO 0.9.0: openapi-core (capa L3a, en tests) exige `<0.9.0`, así que
# subir esta versión deja la validación en tiempo de ejecución y la de runtime en
# ramas incompatibles. 0.8.5 es la última pareja que soporta OpenAPI 3.1 en ambas.
openapi-spec-validator==0.8.5

# === Subida de archivos (multipart/form-data) ===

python-multipart
